cmake_minimum_required(VERSION 3.20)
project(JttyWorkbench VERSION 0.1.0 LANGUAGES C CXX Fortran)
set(CMAKE_Fortran_MODULE_DIRECTORY ${CMAKE_BINARY_DIR}/modules)
set(WSJT ${CMAKE_SOURCE_DIR}/upstream/wsjtx)
find_library(FFTW_FLOAT fftw3f HINTS /opt/homebrew/lib /usr/local/lib REQUIRED)
set(shared packjt77_schema packjt77_grammar packjt77 free_text)
set(sources)
foreach(name IN LISTS shared)
  list(APPEND sources ${WSJT}/lib/77bit/${name}.f90)
endforeach()
add_executable(upstream-sjtty ${WSJT}/lib/jtty/sjtty.f90 ${WSJT}/lib/wavhdr.f90 ${WSJT}/lib/ft8/watterson.f90 ${WSJT}/lib/gran.c)
target_link_libraries(upstream-sjtty PRIVATE jtty_core)
foreach(name fftw3mod four2a chkcall db twkfreq gfsk_pulse)
  list(APPEND sources ${WSJT}/lib/${name}.f90)
endforeach()
foreach(name jtty_source_codec jtty_mod jtty_tbcc_code_profile jtty_tbcc_list_decoder tbcc jtty_tbcc_decoder jtty_payload_correlators jtty_fec_mod jtty_mdecode ana64a jtty_peakup jtty_block_pow gen_syncwave gen_jttywave subtract_jtty)
  list(APPEND sources ${WSJT}/lib/jtty/${name}.f90)
endforeach()
add_library(jtty_core STATIC ${sources})
target_include_directories(jtty_core PRIVATE ${WSJT}/lib)
if(CMAKE_Fortran_COMPILER_ID STREQUAL "GNU")
  target_compile_options(jtty_core PRIVATE -std=legacy -fallow-argument-mismatch -ffree-line-length-none)
endif()
target_link_libraries(jtty_core PUBLIC ${FFTW_FLOAT})
add_executable(jtty-engine native/engine.f90)
target_link_libraries(jtty-engine PRIVATE jtty_core)
find_path(HAMLIB_INCLUDE hamlib/rig.h HINTS /opt/homebrew/include /usr/local/include REQUIRED)
find_library(HAMLIB_LIBRARY hamlib HINTS /opt/homebrew/lib /usr/local/lib REQUIRED)
add_library(jtty_live SHARED native/live_bridge.f90 native/radio.c)
target_include_directories(jtty_live PRIVATE ${HAMLIB_INCLUDE})
target_link_libraries(jtty_live PRIVATE jtty_core ${HAMLIB_LIBRARY} "-framework CoreAudio")
enable_testing()
foreach(name source_codec tbcc_code_profile tbcc_decoder payload_correlators)
  add_executable(test_${name} ${WSJT}/tests/unit/jtty/test_jtty_${name}.f90)
  target_link_libraries(test_${name} PRIVATE jtty_core)
  add_test(NAME ${name} COMMAND test_${name})
endforeach()

list(APPEND CMAKE_PREFIX_PATH /opt/homebrew/opt/qt@5 /usr/local/opt/qt@5)
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Multimedia)
add_executable(jtty-audio-input native/qt-audio/main.cpp
  ${WSJT}/Audio/soundin.cpp ${WSJT}/Audio/soundin.h
  ${WSJT}/Audio/AudioDevice.cpp ${WSJT}/Audio/AudioInputSource.hpp
  ${WSJT}/Audio/AudioStreamDescriptor.cpp)
set_target_properties(jtty-audio-input PROPERTIES AUTOMOC ON CXX_STANDARD 17)
target_include_directories(jtty-audio-input PRIVATE native/qt-audio ${WSJT})
target_link_libraries(jtty-audio-input PRIVATE Qt5::Core Qt5::Multimedia "-framework CoreAudio" "-framework CoreFoundation")
add_test(NAME upstream_audio_adapter COMMAND jtty-audio-input --self-test)
